## 設計

### SkillNode — 包裝 Node 成為可執行的技能

```rust
/// SkillNode — 包裝 Skill，使其符合 Node trait
pub struct SkillNode<S: Skill> {
    skill: S,
    id: String,
}

impl<S: Skill> Node for SkillNode<S> {
    fn id(&self) -> &str { &self.id }
    fn dependencies(&self) -> Vec<&str> { self.skill.dependencies() }
    fn category(&self) -> NodeCategory { NodeCategory::Skill }
    fn execute(&self, ctx: &Context) -> NodeResult {
        let input = ctx.get("input").unwrap_or("{}");
        self.skill.execute(input).into()
    }
    fn as_any(&self) -> &dyn Any { self }
}
```

### Skill Trait — 技能的具體實作

```rust
pub trait Skill: Send + Sync {
    fn id(&self) -> &str;
    fn name(&self) -> &str;
    fn execute(&self, input: &str) -> SkillOutput;
    fn dependencies(&self) -> Vec<&str>; // 此技能依賴哪些其他技能
    fn input_format(&self) -> &str;
    fn output_format(&self) -> &str;
}
```

### Executor 與 SkillRegistry 整合

```rust
impl Executor {
    /// 執行技能（透過 registry）
    pub fn execute_skill(
        &self,
        registry: &SkillRegistry,
        graph: &mut MemoryGraph,
        skill_id: &str,
        input: &str,
    ) -> NodeResult {
        let skill = registry.get(skill_id)?;
        let skill_node = SkillNode::new(skill);
        graph.add_node(skill_node);
        self.execute(graph, skill_id)
    }
}
```
